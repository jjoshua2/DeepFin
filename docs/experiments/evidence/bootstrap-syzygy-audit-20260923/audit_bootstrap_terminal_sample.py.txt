import io,json,time
from pathlib import Path
import chess,zstandard
from chess_anti_engine.tablebase import open_strict_match_tablebase
start=time.monotonic()
tb=open_strict_match_tablebase('/home/josh/projects/chess/data/syzygy_3-4-5:/home/josh/projects/chess/data/syzygy_6',max_pieces=6)
result={'scope':'First raw shard prefix per source run; selected-cohort membership not requalified; not a prevalence estimate','limits':{'seconds':60,'decompressed_bytes_per_run':33554432,'unique_games_per_run':12},'runs':[]}
try:
 for run in ['run03_s3','run06_g10','run07_g10_companion4']:
  path=Path('/home/josh/projects/chess/data/nnue_bootstrap')/run/'w00-00000.jsonl.zst'
  sample={'run':run,'path':str(path),'games':[],'rows_read':0,'bytes_read':0}
  seen=set()
  with path.open('rb') as raw,zstandard.ZstdDecompressor().stream_reader(raw) as reader,io.TextIOWrapper(reader) as lines:
   for line in lines:
    if time.monotonic()-start>60 or sample['bytes_read']>=33554432 or len(seen)>=12: break
    sample['bytes_read']+=len(line.encode()); sample['rows_read']+=1
    row=json.loads(line); gid=row.get('game_id')
    if gid in seen: continue
    seen.add(gid); a=row.get('adjudication'); rec={'game_id':gid,'result_pgn':row.get('result_pgn'),'adjudication':a}
    if a and a.get('fen'):
     board=chess.Board(a['fen']); wdl=tb.probe_wdl(board); dtz=tb.probe_dtz(board)
     rec.update(terminal_halfmove_clock=board.halfmove_clock,raw_syzygy_wdl=wdl,raw_syzygy_dtz=dtz)
     if abs(wdl)==1 or wdl==0:
      rec['rule50_result']='1/2-1/2'
     elif board.halfmove_clock==0:
      rec['rule50_result']='1-0' if ((wdl>0)==(board.turn==chess.WHITE)) else '0-1'
     else: rec['rule50_result']=None
     rec['confirmed_result_mismatch']=rec['rule50_result'] is not None and rec['rule50_result']!=a.get('result')
    sample['games'].append(rec)
  result['runs'].append(sample)
finally: tb.close()
result['elapsed_s']=time.monotonic()-start
Path('/tmp/bootstrap-terminal-sample-20260923.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({'elapsed_s':result['elapsed_s'],'runs':[{'run':r['run'],'rows_read':r['rows_read'],'bytes_read':r['bytes_read'],'games':len(r['games']),'adjudicated':sum(bool(x.get('adjudication')) for x in r['games']),'mismatch':sum(bool(x.get('confirmed_result_mismatch')) for x in r['games'])} for r in result['runs']]},indent=2))
