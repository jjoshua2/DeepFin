# Singleton selection and initialized geometric attack witness composition.
import Base
import ./Spec.bend as S
import ./Index.bend as Index
import ./Flow.bend as Flow
import ./Evidence.bend as E
import ../attack_witness/Queries.bend as WQ
import ../attack_witness/LAWS.bend as Witness
import ../attack_witness/PROOF.bend as WitnessProof
import ../../../legal_probe/Chess.bend as Chess

def actual(+d: Nat,+seed: U64,+n: Nat,+extra: Nat,+b: Chess.Board,+sq: Nat,+side: Bool,
  cert: E.bundle(d,seed,n,extra,b,sq,Bool.not(side)),+bound: {Nat.is_lt(sq,64n) == True{} : Bool},
  one: {U64.and(Chess.get_kings(b),Chess.color(b,Bool.to_u32(side))) == U64.bit(sq) : U64}) ->
  {Chess.in_check(S.init(d,seed,n,extra),b,Bool.to_u32(side)) ==
    (S.init(d,seed,n,extra),S.attacked(b,sq,Bool.not(side))) : Array<U64> & Bool}:
  Equal.trans(Array<U64> & Bool,
    Chess.in_check(S.init(d,seed,n,extra),b,Bool.to_u32(side)),
    WQ.run(S.init(d,seed,n,extra),b,U32.from_nat(sq),Bool.to_u32(Bool.not(side))),
    (S.init(d,seed,n,extra),S.attacked(b,sq,Bool.not(side))),
    Equal.trans(Array<U64> & Bool,
      Chess.in_check(S.init(d,seed,n,extra),b,Bool.to_u32(side)),
      WQ.run(S.init(d,seed,n,extra),b,U32.from_nat(sq),U32.xor(Bool.to_u32(side),1)),
      WQ.run(S.init(d,seed,n,extra),b,U32.from_nat(sq),Bool.to_u32(Bool.not(side))),
      Witness.single_king_in_check_uses_witness_reduction(S.init(d,seed,n,extra),b,Bool.to_u32(side),sq,bound,one),
      Equal.cong(U32,Array<U64> & Bool,by => WQ.run(S.init(d,seed,n,extra),b,U32.from_nat(sq),by),
        U32.xor(Bool.to_u32(side),1),Bool.to_u32(Bool.not(side)),Index.flip(side))),
    Flow.run(d,seed,n,extra,b,sq,Bool.not(side),cert))
