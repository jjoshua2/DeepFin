  {Nat.is_lt(sq,128n) == True{} : Bool} &

   {U32.add(64,U32.from_nat(sq)) == U32.from_nat(Nat.add(64n,sq)) : U32})

def rook_part(+sq: Nat,c: cert(sq)) -> {Nat.is_lt(sq,128n) == True{} : Bool}:
  (r,rest) = c
  r

def bishop_part(+sq: Nat,c: cert(sq)) -> {Nat.is_lt(Nat.add(64n,sq),128n) == True{} : Bool}:
  (r,rest) = c
  (b,cast) = rest
  b

def key_part(+sq: Nat,c: cert(sq)) -> {U32.add(64,U32.from_nat(sq)) == U32.from_nat(Nat.add(64n,sq)) : U32}:
  (r,rest) = c
  (b,cast) = rest
  cast

def rook_bound(+sq: Nat,bound: {Nat.is_lt(sq,64n) == True{} : Bool}) ->
  {Nat.is_lt(sq,128n) == True{} : Bool}:
  rook_part(sq,fixed(sq,bound))

def bishop_bound(+sq: Nat,bound: {Nat.is_lt(sq,64n) == True{} : Bool}) ->
  {Nat.is_lt(Nat.add(64n,sq),128n) == True{} : Bool}:
  bishop_part(sq,fixed(sq,bound))

def bishop_key(+sq: Nat,bound: {Nat.is_lt(sq,64n) == True{} : Bool}) ->
  {U32.add(64,U32.from_nat(sq)) == U32.from_nat(Nat.add(64n,sq)) : U32}:
  key_part(sq,fixed(sq,bound))

def extras_bound(+extra: Nat,full: {extra == 64n : Nat}) ->
  {Nat.is_le(Nat.add(extra,0n),64n) == True{} : Bool}:
  %Equal.sym(Nat,extra,64n,full) : {Nat.is_le(Nat.add(_,0n),64n) == True{} : Bool}
  {==}

def flip(+side: Bool) -> {U32.xor(Bool.to_u32(side),1) == Bool.to_u32(Bool.not(side)) : U32}:
  match side:
    case True{}: {==}
    case False{}: {==}
