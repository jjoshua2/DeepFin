def bishop(+d: Nat,+seed: U64,+n: Nat,+extra: Nat,+b: Chess.Board,+sq: Nat,+by: Bool,+pa: U64,+na: U64,+ka: U64,+ra: U64,
  e4: E.query(d,seed,n,extra,sq,by,Chess.occupied(b),S.Bishop{})) ->
  {WQ.bishop(b,Bool.to_u32(by),pa,na,ka,ra,Chess.attack(2,S.init(d,seed,n,extra),U32.from_nat(sq),Bool.to_u32(by),Chess.occupied(b))) ==
    (S.init(d,seed,n,extra),W.present(b,Bool.to_u32(by),W.Masks{pa,na,ka,ra,S.mask(S.Bishop{},sq,by,Chess.occupied(b))})) : Array<U64> & Bool}:
  Equal.cong(Array<U64> & U64,Array<U64> & Bool,r => WQ.bishop(b,Bool.to_u32(by),pa,na,ka,ra,r),
      Chess.attack(2,S.init(d,seed,n,extra),U32.from_nat(sq),Bool.to_u32(by),Chess.occupied(b)),(S.init(d,seed,n,extra),S.mask(S.Bishop{},sq,by,Chess.occupied(b))),e4)

  e3: E.query(d,seed,n,extra,sq,by,Chess.occupied(b),S.Rook{}),e4: E.query(d,seed,n,extra,sq,by,Chess.occupied(b),S.Bishop{})) ->

    (S.init(d,seed,n,extra),W.present(b,Bool.to_u32(by),W.Masks{pa,na,ka,S.mask(S.Rook{},sq,by,Chess.occupied(b)),S.mask(S.Bishop{},sq,by,Chess.occupied(b))})),
    Equal.cong(Array<U64> & U64,Array<U64> & Bool,r => WQ.rook(b,U32.from_nat(sq),Bool.to_u32(by),pa,na,ka,r),

  {WQ.pawn(b,U32.from_nat(sq),Bool.to_u32(by),Chess.attack(0,S.init(d,seed,n,extra),U32.from_nat(sq),U32.xor(Bool.to_u32(by),1),Chess.occupied(b))) ==
    (S.init(d,seed,n,extra),W.present(b,Bool.to_u32(by),W.Masks{S.mask(S.Pawn{},sq,Bool.not(by),Chess.occupied(b)),S.mask(S.Knight{},sq,by,Chess.occupied(b)),S.mask(S.King{},sq,by,Chess.occupied(b)),S.mask(S.Rook{},sq,by,Chess.occupied(b)),S.mask(S.Bishop{},sq,by,Chess.occupied(b))})) : Array<U64> & Bool}:

    WQ.pawn(b,U32.from_nat(sq),Bool.to_u32(by),Chess.attack(0,S.init(d,seed,n,extra),U32.from_nat(sq),U32.xor(Bool.to_u32(by),1),Chess.occupied(b))),

    knight(d,seed,n,extra,b,sq,by,S.mask(S.Pawn{},sq,Bool.not(by),Chess.occupied(b)),e1,e2,e3,e4))

def run(+d: Nat,+seed: U64,+n: Nat,+extra: Nat,+b: Chess.Board,+sq: Nat,+by: Bool,cert: E.bundle(d,seed,n,extra,b,sq,by)) ->
  {WQ.run(S.init(d,seed,n,extra),b,U32.from_nat(sq),Bool.to_u32(by)) == (S.init(d,seed,n,extra),S.attacked(b,sq,by)) : Array<U64> & Bool}:
  (e0,rest1) = cert
  (e1,rest2) = rest1
  (e2,rest3) = rest2
  (e3,e4) = rest3
  pawn(d,seed,n,extra,b,sq,by,e0,e1,e2,e3,e4)

def attacked(+d: Nat,+seed: U64,+n: Nat,+extra: Nat,+b: Chess.Board,+sq: Nat,+by: Bool,cert: E.bundle(d,seed,n,extra,b,sq,by)) ->
  {Chess.attacked(S.init(d,seed,n,extra),b,U32.from_nat(sq),Bool.to_u32(by)) == (S.init(d,seed,n,extra),S.attacked(b,sq,by)) : Array<U64> & Bool}:
  Equal.trans(Array<U64> & Bool,
    Chess.attacked(S.init(d,seed,n,extra),b,U32.from_nat(sq),Bool.to_u32(by)),WQ.run(S.init(d,seed,n,extra),b,U32.from_nat(sq),Bool.to_u32(by)),
    (S.init(d,seed,n,extra),S.attacked(b,sq,by)),Witness.actual_attacked_uses_witness_reduction(S.init(d,seed,n,extra),b,U32.from_nat(sq),Bool.to_u32(by)),
    run(d,seed,n,extra,b,sq,by,cert))
