import Base
import ./PROOF.bend as Proof
import ./LAWS.bend as Laws
import ./Bind.bend as Bind
import ../../../legal_probe/Chess.bend as Chess
import ../initialized_attacks/Build.bend as Init

def scan(table: Array<U64>,+keys: List<&2,U32>,+b: Chess.Board,+tail: List<&2,Chess.Ply>) ->
  {Bind.moves_table(Chess.scan(keys,b,(table,tail))) == table : Array<U64>}:
  Laws.ordinary_scan_preserves_table(table,keys,b,tail)

def filter(table: Array<U64>,+b: Chess.Board,+moves: List<&2,Chess.Ply>) ->
  {Bind.moves_table(Chess.filter_prepare(b,(table,moves))) == table : Array<U64>}:
  Laws.optimized_filter_preserves_table(table,b,moves)

def legal(table: Array<U64>,+b: Chess.Board) ->
  {Bind.moves_table(Chess.legal_moves(table,b)) == table : Array<U64>}:
  Laws.complete_generator_preserves_table(table,b)

# No depth/count/seed correctness premise is required for this storage invariant.
def initialized(+d: Nat,+seed: U64,+n: Nat,+extra: Nat,+b: Chess.Board) ->
  {Bind.moves_table(Chess.legal_moves(Init.run(d,seed,n,extra),b)) == Init.run(d,seed,n,extra) : Array<U64>}:
  Laws.complete_generator_preserves_table(Init.run(d,seed,n,extra),b)
