# No proof modules or reference-produced moves/masks are imported.
import Base

def show(+x: U64) -> String:
  U32.show(U64.high(x)) ++ " " ++ U32.show(U64.low(x))

def word(r: Array<U64> & U64) -> IO(Array<U64>):
  (table,v) = r
  do IO<Array<U64>>:
    IO.print("word " ++ show(v))
    return table

def dump(n: Nat,+i: U32,table: Array<U64>) -> IO(Array<U64>):
  match n:
    case 0n: IO.pure(Array<U64>,table)
    case 1n+p:
      do IO<Array<U64>>:
        table : Array<U64> <- word(Array.get(U64,table,i))
        dump(p,U32.inc(i),table)

def snapshot(+n: Nat,label: String,table: Array<U64>) -> IO(Array<U64>):
  do IO<Array<U64>>:
    IO.print("state " ++ label ++ " " ++ U32.show(U32.from_nat(n)))
    table : Array<U64> <- dump(n,0,table)
    IO.print("end-state")
    return table

def moves(ms: List<&2,Chess.Ply>) -> IO(Unit):
  match ms:
    case Nil{}: IO.print("end-moves")
    case Con{Chess.Ply{src,dst,promotion,flag},rest}:
      do IO<Unit>:
        IO.print("move " ++ U32.show(src) ++ " " ++ U32.show(dst) ++ " " ++ U32.show(promotion) ++ " " ++ U32.show(flag))
        moves(rest)

def observe(n: Nat,+label: String,r: Array<U64> & List<&2,Chess.Ply>) -> IO(Array<U64> & List<&2,Chess.Ply>):
  (table,+ms) = r
  do IO<Array<U64> & List<&2,Chess.Ply>>:
    IO.print("moves " ++ label ++ " " ++ U32.show(U32.from_nat(List.length(&2,Chess.Ply,ms))))
    moves(ms)
    table : Array<U64> <- snapshot(n,label,table)
    return (table,ms)

def tail(mode: U32) -> List<&2,Chess.Ply>:
  match mode:
    case 0: Nil{}
    case 1: Con{Chess.Ply{17,25,0,0},Nil{}}
    case _: Con{Chess.Ply{4,6,0,2},Con{Chess.Ply{4,6,0,2},Nil{}}}

def finish(r: Array<U64> & List<&2,Chess.Ply>) -> IO(Array<U64>):
  (table,ms) = r
  do IO<Array<U64>>:
    IO.print("end-case")
    return table

def after_filter(+n: Nat,+b: Chess.Board,r: Array<U64> & List<&2,Chess.Ply>) -> IO(Array<U64>):
  (table,ms) = r
  do IO<Array<U64>>:
    r : Array<U64> & List<&2,Chess.Ply> <- observe(n,"legal",Chess.legal_moves(table,b))
    finish(r)

def run_case(+n: Nat,mode: U32,+b: Chess.Board,table: Array<U64>) -> IO(Array<U64>):
  +own = Chess.color(b,Chess.get_turn(b))
  do IO<Array<U64>>:
    IO.print("case")
    r : Array<U64> & List<&2,Chess.Ply> <- observe(n,"scan",Chess.scan(Chess.bit_squares(64n,U64.is_zero(own),own,Nil{}),b,(table,tail(mode))))
    r : Array<U64> & List<&2,Chess.Ply> <- observe(n,"castles",Chess.castle_side(b,False{},Chess.castle_side(b,True{},r)))
    r : Array<U64> & List<&2,Chess.Ply> <- observe(n,"filter",Chess.filter_prepare(b,r))
    after_filter(n,b,r)

def checked(ok: Bool,+n: Nat,mode: U32,b: Chess.Board,table: Array<U64>) -> IO(Array<U64>):
  match ok:
    case True{}: run_case(n,mode,b,table)
    case False{}: IO.die(Array<U64>,2,"invalid preservation request")

def rows(fuel: Nat,+n: Nat,xs: List<&2,U32>,table: Array<U64>) -> IO(Unit):
  match fuel xs:
    case _ Nil{}: IO.pure(Unit,Unit{})
    case 1n+p +mode <> ph <> pl <> nh <> nl <> bh <> bl <> rh <> rl <> qh <> ql <> kh <> kl <> wh <> wl <> blh <> bll <> turn <> rights <> ep <> rest:
      do IO<Unit>:
        table : Array<U64> <- checked(U32.is_le(mode,2),n,mode,Chess.Board{U64.from_parts(ph,pl),U64.from_parts(nh,nl),U64.from_parts(bh,bl),U64.from_parts(rh,rl),U64.from_parts(qh,ql),U64.from_parts(kh,kl),U64.from_parts(wh,wl),U64.from_parts(blh,bll),turn,rights,ep},table)
        rows(p,n,rest,table)
    case _ _: IO.die(Unit,2,"invalid preservation tuple or budget")

def fill(n: Nat,table: Array<U64>) -> Array<U64>:
  match n:
    case 0n: table
    case 1n+p:
      +p = p
      +i = U32.from_nat(p)
      fill(p,Array.set(U64,table,i,U64.from_parts(U32.add(305419896,i),U32.xor(2596069104,U32.mul(i,2654435761)))))

def start(+n: Nat,table: Array<U64>,xs: List<&2,U32>) -> IO(Unit):
  do IO<Unit>:
    table : Array<U64> <- snapshot(n,"initial",table)
    rows(8n,n,xs,table)

def context(mode: U32,xs: List<&2,U32>) -> IO(Unit):
  match mode:
    case 0: start(1n,ALeaf{U64.from_parts(305419896,2596069104)},xs)
    case 1: start(8n,fill(8n,Array.new(U64,3n,U64.zero())),xs)
    case 2: start(512n,fill(512n,Array.new(U64,9n,U64.zero())),xs)
    case 3: start(131072n,Array.set(U64,Tables.build(),131071,U64.from_parts(826366246,1398314899)),xs)
    case 4: start(131072n,fill(131072n,Array.new(U64,17n,U64.zero())),xs)
    case _: IO.die(Unit,2,"invalid preservation context")

def numbers(xs: List<String>,acc: List<&2,U32>) -> Maybe<&2,List<&2,U32>>:
  match xs:
    case Nil{}: Some{List.reverse(&2,U32,acc)}
    case h <> t:
      do Maybe<&2,List<&2,U32>>:
        v : U32 <- Text.number(h)
        numbers(t,v <> acc)

def parsed(xs: Maybe<&2,List<&2,U32>>) -> IO(Unit):
  match xs:
    case Some{mode <> rows}: context(mode,rows)
    case _: IO.die(Unit,2,"invalid preservation arguments")

def main() -> IO(Unit):
  do IO<Unit>:
    xs : List<String> <- IO.args()
    parsed(numbers(xs,Nil{}))
