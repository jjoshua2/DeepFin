class AdapterError(RuntimeError):
    pass


def sha(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


def file_sha(path: Path) -> str:
    return sha(path.read_bytes())


def verify_ref(path_text: str, expected: str, label: str) -> Path:
    path = Path(path_text).resolve(strict=True)
    if len(expected) != 64 or file_sha(path) != expected:
        raise AdapterError(f"{label} SHA256 mismatch")
    return path


def atomic_json(path: Path, value: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    raw = (json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False) + "\n").encode()
    temp = path.with_name(path.name + f".tmp.{os.getpid()}")
    with temp.open("xb") as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())
    os.replace(temp, path)
    fd = os.open(path.parent, os.O_RDONLY | os.O_DIRECTORY)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)


def checked_overlay() -> dict[str, dict[str, Any]]:
    result: dict[str, dict[str, Any]] = {}
    for role, path in OVERLAY_FILES.items():
        if path.is_symlink() or not path.is_file():
            raise AdapterError(f"overlay source missing or symlinked: {role}")
        result[role] = {"path": str(path.resolve()), "sha256": file_sha(path), "bytes": path.stat().st_size}
    if file_sha(OVERLAY_FILES["versioned_durable"]) != file_sha(OVERLAY_FILES["versioned_durable_package_copy"]):
        raise AdapterError("durable module copies differ")
    return result


def verify_qualified_source() -> None:
    head = subprocess.run(["git", "rev-parse", "HEAD"], cwd=QUALIFIED_ROOT,
                          check=True, capture_output=True, text=True).stdout.strip()
    dirty = subprocess.run(["git", "status", "--porcelain", "--untracked-files=no"], cwd=QUALIFIED_ROOT,
                           check=True, capture_output=True, text=True).stdout
    if head != QUALIFIED_HEAD or dirty:
        raise AdapterError("qualified arena source head or tracked cleanliness differs")
    source_runner = QUALIFIED_ROOT / "scripts/arena_standard.py"
    source_durable = QUALIFIED_ROOT / "chess_anti_engine/eval/arena_durable.py"
    if file_sha(source_runner) != FROZEN_RUNNER_SHA or file_sha(source_durable) != FROZEN_DURABLE_SHA:
        raise AdapterError("qualified frozen arena source hashes differ")


def _ref_object(value: Any, label: str) -> tuple[str, str]:
    if not isinstance(value, dict) or set(value) != {"path", "sha256"}:
        raise AdapterError(f"{label} ref must contain exactly path and sha256")
    return str(Path(value["path"]).resolve()), value["sha256"]


def _linux_start_ticks(pid: int) -> str:
    stat = Path(f"/proc/{pid}/stat").read_text()
    tail = stat[stat.rfind(")") + 2:].split()
    return tail[19]


def verify_authority(args: argparse.Namespace, refs: dict[str, Path]) -> None:
    max_bytes = 8 * 1024 * 1024
    docs = {}
    for role in ("operation", "authorization", "budget", "lane_release", "operation_baseline"):
        path = refs[role]
        if path.stat().st_size > max_bytes:
            raise AdapterError(f"{role} metadata exceeds 8 MiB")
        docs[role] = json.loads(path.read_bytes())
    operation = docs["operation"]
    authorization = docs["authorization"]
    lane = docs["lane_release"]
    arena = operation.get("arena")
    if not isinstance(arena, dict):
        raise AdapterError("operation arena contract missing")
    adapter_ref = _ref_object(arena.get("adapter_ref"), "adapter source")
    if adapter_ref != (str(Path(__file__).resolve()), file_sha(Path(__file__))):
        raise AdapterError("operation adapter source path/SHA differs")
    catalog_ref = _ref_object(arena.get("tablebase_catalog_ref"), "tablebase catalog")
    if catalog_ref != (str(Path(args.tablebase_catalog).resolve()), file_sha(Path(args.tablebase_catalog).resolve(strict=True))):
        raise AdapterError("operation tablebase catalog path/SHA differs")
    config_ref = _ref_object(arena.get("live_config_ref"), "live config")
    if config_ref != (str(FROZEN_CONFIG.resolve()), file_sha(FROZEN_CONFIG)):
        raise AdapterError("operation live config path/SHA differs")
    expected_caps = {"verification_seconds": 60, "max_sessions": 25,
                     "whole_attempt_seconds": 1650, "phase_wall_seconds": 43200,
                     "pairs": 2606, "games": 5212}
    if any(arena.get(key) != value for key, value in expected_caps.items()):
        raise AdapterError("operation arena phase/count caps differ")
    baseline = docs["operation_baseline"]
    if lane.get("schema") != "deepfin_three_pass_runtime_lane_release_v1" or lane.get("status") != "PASS_PARENT_REVIEWED_THREE_PASS_LANE_RELEASE":
        raise AdapterError("normalized lane release schema/status invalid")
    if lane.get("owner_stop_confirmed") is not True or not isinstance(lane.get("nonce"), str) or not lane["nonce"]:
        raise AdapterError("lane release lacks confirmed owner stop or nonce")
    for field, role, digest in (("operation_ref", "operation", args.operation_sha256),
                                ("whole_operation_authorization_ref", "authorization", args.authorization_sha256)):
        if _ref_object(lane.get(field), field) != (str(refs[role]), digest):
            raise AdapterError(f"lane release {field} does not bind supplied ref")
    operation_budget = _ref_object(operation.get("budget_ref"), "operation budget")
    if operation_budget != (str(refs["budget"]), args.budget_sha256):
        raise AdapterError("operation budget ref differs from supplied budget")
    if authorization.get("budget_sha256") != args.budget_sha256:
        raise AdapterError("whole-operation authorization budget pin differs")
    if authorization.get("operation_sha256") != args.operation_sha256:
        raise AdapterError("whole-operation authorization operation pin differs")
    if baseline.get("schema") != "deepfin_three_pass_science_slot_baseline_v1":
        raise AdapterError("whole-operation baseline schema invalid")
    if (baseline.get("operation_sha256") != args.operation_sha256
            or baseline.get("authorization_sha256") != args.authorization_sha256):
        raise AdapterError("whole-operation baseline pins differ")
    lane_ref = baseline.get("lane_release_ref")
    if _ref_object(lane_ref, "baseline lane release") != (str(refs["lane_release"]), args.lane_release_sha256):
        raise AdapterError("baseline lane release ref differs")
    owner = lane.get("science_owner_identity")
    baseline_owner = baseline.get("owner_identity")
    if not isinstance(owner, dict) or not isinstance(baseline_owner, dict):
        raise AdapterError("lane/baseline owner identity missing")
    identity = {key: str(owner.get(key)) for key in ("pid", "start_ticks", "boot_id")}
    baseline_identity = {key: str(baseline_owner.get(key)) for key in ("pid", "start_ticks", "boot_id")}
    if identity != baseline_identity:
        raise AdapterError("lane owner differs from operation baseline owner")
    pid = int(identity["pid"])
    if os.getppid() != pid:
        raise AdapterError("current adapter parent is not the reviewed science owner")
    if _linux_start_ticks(pid) != identity["start_ticks"]:
        raise AdapterError("owner PID start ticks changed")
    if Path("/proc/sys/kernel/random/boot_id").read_text().strip() != identity["boot_id"]:
        raise AdapterError("owner boot ID changed")
    now = __import__("time").monotonic()
    if not (float(baseline.get("scope_started_monotonic", now + 1)) <= now
            < float(baseline.get("scope_deadline_monotonic", now))):
        raise AdapterError("whole-operation scope deadline is invalid or expired")
    lease_identities = lane.get("canonical_lease_identities")
    expected = {
        "GPU": (2096, 81348917, os.environ.get("DEEPFIN_INHERITED_GPU_LOCK_FD")),
        "IO": (2096, 18503149, os.environ.get("DEEPFIN_INHERITED_IO_LOCK_FD")),
    }
    if not isinstance(lease_identities, dict):
        raise AdapterError("lane release canonical lease identities missing")
    for role, (dev, ino, fd_raw) in expected.items():
        matching = [(Path(k), v) for k, v in lease_identities.items()
                    if v == [dev, ino]]
        if len(matching) != 1:
            raise AdapterError(f"lane release {role} lease identity differs")
        path, identity_record = matching[0]
        if fd_raw is None or not fd_raw.isdecimal():
            raise AdapterError(f"inherited {role} lock fd missing")
        fd = int(fd_raw)
        st = os.fstat(fd)
        if (st.st_dev, st.st_ino) != (dev, ino):
            raise AdapterError(f"inherited {role} fd does not refer to canonical lease")
        if fcntl.fcntl(fd, fcntl.F_GETFL) & os.O_ACCMODE != os.O_RDONLY:
            raise AdapterError(f"inherited {role} fd is not read-only")
        path_stat = path.stat()
        if (path_stat.st_dev, path_stat.st_ino) != (dev, ino):
            raise AdapterError(f"canonical {role} lease path identity changed")


def validate_request_files(args: argparse.Namespace) -> dict[str, Path]:
    refs = {
        "operation": verify_ref(args.operation, args.operation_sha256, "operation"),
        "authorization": verify_ref(args.authorization, args.authorization_sha256, "authorization"),
        "budget": verify_ref(args.budget, args.budget_sha256, "budget"),
        "lane_release": verify_ref(args.lane_release, args.lane_release_sha256, "lane release"),
        "operation_baseline": verify_ref(args.operation_baseline, args.operation_baseline_sha256, "operation baseline"),
    }
    verify_authority(args, refs)
    # Only after owner identity and inherited canonical FDs are verified do we
    # read/hash the panel or checkpoint payloads.
    refs.update({
        "panel": verify_ref(args.panel_path, args.panel_sha256, "panel"),
        "control": verify_ref(args.control_path, args.control_sha256, "control checkpoint"),
        "candidate": verify_ref(args.candidate_path, args.candidate_sha256, "candidate checkpoint"),
    })
    return refs

def load_versioned_durable():
    verify_qualified_source()
    overlay = str(PACKET)
    scripts = str(PACKET / "scripts")
    qualified = str(QUALIFIED_ROOT)
    usersite = "/home/josh/.local/lib/python3.10/site-packages"
    for item in (usersite, qualified, overlay, scripts):
        if item not in sys.path:
            sys.path.insert(0, item)
    spec = importlib.util.spec_from_file_location("arena_durable_three_pass", OVERLAY_FILES["versioned_durable"])
    if spec is None or spec.loader is None:
        raise AdapterError("could not load pinned durable module")
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    if Path(module.__file__).resolve() != OVERLAY_FILES["versioned_durable"].resolve():
        raise AdapterError("durable module origin mismatch")
    return module

def verify_manifest(args: argparse.Namespace) -> tuple[dict[str, Any], dict[str, Path]]:
    refs = validate_request_files(args)
    manifest_path = verify_ref(args.manifest_path, args.manifest_sha256, "prepared manifest")
    manifest = json.loads(manifest_path.read_bytes())
    if manifest.get("schema") != "deepfin_three_pass_arena_prepared_v1" or manifest.get("status") != "PASS_PREPARED_THREE_PASS_ARENA":
        raise AdapterError("prepared manifest status/schema invalid")
    for k, path_key, digest_key in (("panel", "panel_path", "panel_sha256"),
                                    ("control", "control_path", "control_sha256"),
                                    ("candidate", "candidate_path", "candidate_sha256"),
                                    ("operation", "operation", "operation_sha256"),
                                    ("authorization", "authorization", "authorization_sha256"),
                                    ("budget", "budget", "budget_sha256"),
                                    ("lane_release", "lane_release", "lane_release_sha256"),
                                    ("operation_baseline", "operation_baseline", "operation_baseline_sha256")):
        if manifest.get(path_key) != str(refs[k]) or manifest.get(digest_key) != getattr(args, digest_key):
            raise AdapterError(f"prepared manifest {k} binding differs")
    seal = verify_ref(manifest["source_seal_path"], manifest["source_seal_sha256"], "source seal")
    relation_ref=manifest['inference_relation_ref']
    relation=json.loads(verify_ref(relation_ref['path'],relation_ref['sha256'],'inference relation').read_bytes())
    for arm in ('control','candidate'):
        if relation['arms'][arm]['source_ref']!=manifest['original_checkpoint_refs'][arm] or relation['arms'][arm]['export_ref']!={'path':manifest[arm+'_path'],'sha256':manifest[arm+'_sha256']}:
            raise AdapterError('sealed inference relation changed')
    return manifest, refs | {"manifest": manifest_path, "seal": seal}

    env["PYTHONUSERBASE"] = "/home/josh/.local"

def inherited_fds() -> tuple[int, ...]:
    names = ("DEEPFIN_INHERITED_GPU_LOCK_FD", "DEEPFIN_INHERITED_IO_LOCK_FD")
    values = []
    for name in names:
        raw = os.environ.get(name)
        if raw is None or not raw.isdecimal():
            raise AdapterError(f"whole-owner inherited lease fd missing: {name}")
        values.append(int(raw))
    return tuple(values)


def run_child(command: list[str], *, args: argparse.Namespace, output_root: Path,
              result_path: Path, mode: str) -> int:
    fds = inherited_fds()
    env = set_runtime_env(args, output_root)
    log_dir = output_root / "adapter-logs"
    log_dir.mkdir(parents=True, exist_ok=True)
    label = f"attempt-{args.attempt_number:02d}" if mode == "attempt" else mode
    stdout_path, stderr_path = log_dir / f"{label}.stdout", log_dir / f"{label}.stderr"
    if result_path.exists() or stdout_path.exists() or stderr_path.exists():
        raise AdapterError(f"immutable child output already exists: {label}")
    with stdout_path.open("xb") as stdout, stderr_path.open("xb") as stderr:
        child = subprocess.run(command, cwd=QUALIFIED_ROOT, env=env,
                               pass_fds=fds, stdout=stdout, stderr=stderr,
                               check=False)
    manifest = json.loads(Path(args.manifest_path).read_bytes())
    result = {"schema": "deepfin_three_pass_arena_child_result_v1",
              "status": ("PASS_ATTEMPT_EXIT_0" if child.returncode == 0
                         else "VALID_INCOMPLETE_ATTEMPT_EXIT_3" if child.returncode == 3
                         else "HOLD_ATTEMPT_EXIT"),
              "mode": mode, "returncode": child.returncode,
              "manifest_ref": str(Path(args.manifest_path).resolve()),
              "manifest_sha256": args.manifest_sha256,
              "source_seal_ref": manifest["source_seal_path"],
              "source_seal_sha256": manifest["source_seal_sha256"],
              "panel_sha256": manifest["panel_sha256"],
              "control_sha256": manifest["control_sha256"],
              "candidate_sha256": manifest["candidate_sha256"],
              "stdout_path": str(stdout_path), "stderr_path": str(stderr_path)}
    atomic_json(result_path, result)
    print(json.dumps(result, sort_keys=True))
    return child.returncode


def mode_attempt(args: argparse.Namespace) -> int:
    manifest, refs = verify_manifest(args)
    output_root = Path(args.output_root).resolve(strict=True)
    if checked_overlay() != manifest.get("overlay_files"):
        raise AdapterError("overlay files changed after prepare")
    verify_qualified_source()
    command = [sys.executable, "-B", "/mnt/c/Users/jjosh/Documents/Codex/2026-10-08/task/short_e_pair_bound_recovery_v1.py", *manifest["attempt_argv"]]
    # Runner has an explicit pinned sys.path bootstrap at top-level. Child inherits
    # both canonical lock descriptors; whole-owner controls its process group/deadline.
    if args.attempt_number is None or args.attempt_number < 1 or args.attempt_number > 25:
        raise AdapterError("attempt mode requires an attempt number in 1..25")
    return run_child(command, args=args, output_root=output_root,
                     result_path=output_root / f"attempt-{args.attempt_number:02d}-result.json",
                     mode="attempt")


def mode_finalize(args: argparse.Namespace, mode: str) -> int:
    manifest, refs = verify_manifest(args)
    output_root = Path(args.output_root).resolve(strict=True)
    if checked_overlay() != manifest.get("overlay_files"):
        raise AdapterError("overlay files changed after prepare")
    output = output_root / "durable"
    command = [sys.executable, "-B", str(OVERLAY_FILES["versioned_finalizer"]),
        "--mode", mode, "--panel", str(refs["panel"]), "--panel-sha256", args.panel_sha256,
        "--seal", str(refs["seal"]), "--seal-sha256", manifest["source_seal_sha256"],
        "--games-out", str(output / "games.jsonl"), "--pgn-out", str(output / "games.pgn"),
        "--pair-receipts-dir", str(output / "pair-receipts"), "--control-path", str(refs["control"]),
        "--control-sha256", args.control_sha256, "--candidate-path", str(refs["candidate"]),
        "--candidate-sha256", args.candidate_sha256, "--output-root", str(output)]
    return run_finalizer(command, args=args, output_root=output_root, mode=mode,
                         manifest=manifest, refs=refs)


def run_finalizer(command: list[str], *, args: argparse.Namespace, output_root: Path,
                  mode: str, manifest: dict[str, Any], refs: dict[str, Path]) -> int:
    fds = inherited_fds()
    env = set_runtime_env(args, output_root)
    child = subprocess.run(command, cwd=QUALIFIED_ROOT, env=env, pass_fds=fds,
                           capture_output=True, text=True, check=False)
    if child.returncode != 0:
        raise AdapterError(f"{mode} readback refused; returncode={child.returncode}; stderr={child.stderr[-4000:]}")
    try:
        result = json.loads(child.stdout)
    except json.JSONDecodeError as exc:
        raise AdapterError(f"{mode} readback did not emit one JSON object") from exc
    expected_status = "VALID_INCOMPLETE" if mode == "progress" else "PASS_THREE_PASS_2606_COMPLETE"
    if result.get("status") != expected_status:
        raise AdapterError(f"{mode} readback returned unexpected status")
    if result.get("source_seal_sha256") != manifest["source_seal_sha256"]:
        raise AdapterError(f"{mode} readback source seal binding mismatch")
    result["seal_sha256"] = result["source_seal_sha256"]
    result["control_sha256"] = args.control_sha256
    result["candidate_sha256"] = args.candidate_sha256
    pairs, games = result.get("pairs"), result.get("games")
    if type(pairs) is not int or type(games) is not int or games != 2 * pairs:
        raise AdapterError(f"{mode} readback count consistency failed")
    for key, expected in (("panel_sha256", args.panel_sha256),
                          ("seal_sha256", manifest["source_seal_sha256"]),
                          ("control_sha256", args.control_sha256),
                          ("candidate_sha256", args.candidate_sha256)):
        if result.get(key) != expected:
            raise AdapterError(f"{mode} readback binding mismatch: {key}")
    if mode == "final" and (pairs != 2606 or games != 5212
                            or result.get("all_color_halves_verified") is not True):
        raise AdapterError("final readback incomplete")
    print(json.dumps(result, sort_keys=True))
    return 0


def parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser()
    p.add_argument("--mode", choices=("prepare", "attempt", "progress", "final"), required=True)
    for name in ("operation", "authorization", "budget", "lane-release", "operation-baseline"):
        p.add_argument("--" + name, required=True)
        p.add_argument("--" + name + "-sha256", required=True)
    p.add_argument("--control-path", required=True); p.add_argument("--control-sha256", required=True)
    p.add_argument("--candidate-path", required=True); p.add_argument("--candidate-sha256", required=True)
    p.add_argument("--panel-path", required=True); p.add_argument("--panel-sha256", required=True)
    p.add_argument("--tablebase-catalog", required=True)
    p.add_argument("--output-root", required=True)
    p.add_argument("--manifest-path", required=True)
    p.add_argument("--manifest-sha256")
    p.add_argument("--attempt-number", type=int)
    return p


def main() -> int:
    args = parser().parse_args()
    if args.mode != "prepare" and not args.manifest_sha256:
        raise SystemExit("attempt/progress/final require --manifest-sha256")
    if args.mode == "prepare": return mode_prepare(args)
    if args.mode == "attempt": return mode_attempt(args)
    if args.mode == "progress": return mode_finalize(args, "progress")
    return mode_finalize(args, "final")


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except AdapterError as exc:
        raise SystemExit(f"arena phase adapter refused: {exc}") from exc
