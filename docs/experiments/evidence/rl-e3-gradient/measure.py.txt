def require(value, message):
    if not value:
        raise ValueError(message)


def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1048576), b''):
            h.update(block)
    return h.hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), allow_nan=False).encode()


def checked(ref):
    require(set(ref) == {'path', 'sha256'} and 'UNBOUND' not in ref['path'], 'UNBOUND identity')
    require(sha(ref['path']) == ref['sha256'], 'identity changed: ' + ref['path'])
    return json.loads(Path(ref['path']).read_text())


def write(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, indent=2, sort_keys=True, allow_nan=False)
        stream.write('\n'); stream.flush(); os.fsync(stream.fileno())

def captured_loss(function, output, batch, kwargs):
    """Read the actual frozen source locals, never substitute loss/mask formulas."""
    require(sys.getprofile() is None, 'existing profiler must not be overwritten')
    captured = []
    names = ('sf_wdl_frac_f', 'search_wdl_frac_f', 'game_frac', 'sf_available',
             'search_available', 'sf_effective', 'sf_component', 'search_component',
             'target', 'terminal_taper')
    def profile(frame, event, _arg):
        if event == 'return' and frame.f_code is function.__code__:
            local = frame.f_locals
            require(all(k in local for k in names), 'actual fallback capture field missing')
            captured.append((local['weighted_terms'], {k: local[k] for k in names},
                             local['report_exact_masked_sums']))
    sys.setprofile(profile)
    try:
        result = function(output, batch, **kwargs)
    finally:
        sys.setprofile(None)
    require(len(captured) == 1 and captured[0][2] is False, 'wrong loss capture/normalization')
    terms, fallback, _ = captured[0]
    import torch
    fold = None
    for _name, weight, mean, _mask in terms:
        if weight:
            fold = weight * mean if fold is None else fold + weight * mean
    require(fold is not None and torch.equal(fold, result['total']), 'source ordered total differs')
    require(all(torch.isfinite(mean).all() for _n, w, mean, _m in terms if w), 'active nonfinite loss')
    returned = {'policy': 'policy_ce', 'soft_policy': 'soft_policy_ce', 'future_policy': 'future_policy_ce',
                'wdl': 'blended_wdl_ce', 'categorical': 'categorical_ce',
                'volatility': 'volatility', 'moves_left': 'moves_left'}
    for name, weight, mean, _mask in terms:
        if weight:
            require(name in returned and result[returned[name]] is mean, 'unmapped/nonreturned active component')
    common = {'policy', 'soft_policy', 'future_policy'}
    def group(names):
        values = [w * t for name, w, t, _mask in terms if w and name in names]
        require(values, 'empty active family')
        value = values[0]
        for term in values[1:]: value = value + term
        return value
    active = {n for n, w, _t, _m in terms if w}
    families = {'P': group(common), 'U': group(active - common - {'wdl', 'categorical'}),
                'W': group({'wdl'}), 'C': group({'categorical'})}
    return result, terms, fallback, families


def gradients(value, parameters, torch, np, counter, *, retain):
    counter['gradients'] += 1
    require(counter['gradients'] <= 600, 'extra gradient call')
    if not value.requires_grad:
        vector = np.zeros(sum(p.numel() for p in parameters), dtype=np.float32)
        return vector, {'none':list(range(len(parameters))), 'zero_connected':[],
                        'constant_loss_without_graph':True,
                        'sha256':hashlib.sha256(vector.tobytes()).hexdigest()}
    counter['autograd_calls'] += 1
    values = torch.autograd.grad(value, parameters, retain_graph=retain, allow_unused=True)
    vector = np.empty(sum(p.numel() for p in parameters), dtype=np.float32)
    none, zero, offset = [], [], 0
    for index, (parameter, grad) in enumerate(zip(parameters, values)):
        stop = offset + parameter.numel()
        if grad is None:
            vector[offset:stop] = 0; none.append(index)
        else:
            require(grad.dtype == torch.float32 and bool(torch.isfinite(grad).all()), 'bad trunk gradient')
            part = grad.detach().cpu().contiguous().numpy().reshape(-1)
            vector[offset:stop] = part
            if not np.any(part): zero.append(index)
        offset = stop
    return vector, {'none': none, 'zero_connected': zero,
                    'constant_loss_without_graph':False,
                    'sha256': hashlib.sha256(vector.tobytes()).hexdigest()}


def gram(vectors, np):
    require(len(vectors) == 6, 'wrong family matrix')
    result = np.zeros((6, 6), dtype=np.float64)
    for start in range(0, len(vectors[0]), 65536):
        block = np.stack([v[start:start + 65536] for v in vectors]).astype(np.float64)
        result += block @ block.T
    require(np.isfinite(result).all(), 'nonfinite sufficient statistics')
    return result


def cosine(matrix, left, right):
    def dot(a, b):
        return math.fsum(float(matrix[i, j]) * x * y for i, x in a.items() for j, y in b.items())
    aa, bb = dot(left, left), dot(right, right)
    if aa <= 1e-24 or bb <= 1e-24: return None
    return dot(left, right) / math.sqrt(aa * bb)


def cosines(matrix):
    result = {label: cosine(matrix, {0: 1}, vector) for label, vector in
              [('P_W_A', {2: 1}), ('P_C_A', {3: 1}), ('P_WC_A', {2: 1, 3: 1}),
               ('P_W_B', {4: 1}), ('P_C_B', {5: 1}), ('P_WC_B', {4: 1, 5: 1}), ('P_U', {1: 1})]}
    result['d'] = (result['P_WC_B'] - result['P_WC_A']
                   if result['P_WC_A'] is not None and result['P_WC_B'] is not None else None)
    return result


def relative_error(a, b, np):
    delta = aa = bb = 0.0
    for start in range(0, len(a), 65536):
        x = a[start:start + 65536].astype(np.float64); y = b[start:start + 65536].astype(np.float64)
        delta += float((x - y) @ (x - y)); aa += float(x @ x); bb += float(y @ y)
    return math.sqrt(delta) / max(math.sqrt(aa), math.sqrt(bb), 1e-12)


def measure_row(trainer, arrays, categories, index, kwargs, parameters, mapping, counter, observer, total_control):
    import torch, numpy as np
    from chess_anti_engine.train.losses import compute_loss
    from scripts.offline_replay_epoch import _slice_arrays
    hosts, batches = {}, {}
    for arm in ('A', 'B'):
        raw = _slice_arrays(arrays, np.asarray([index], np.int64))
        raw['categorical_target'] = categories[arm][[index]].copy()
        host = trainer._prepare_host_arrays(raw, rng=np.random.default_rng(0), mirror_prob=0,
                                            rebuild_sf_targets=False)
        hosts[arm] = observer.arrays_signature(host)
        batches[arm] = trainer._host_batch_to_tensors(host)
    counter['forwards'] += 1; require(counter['forwards'] <= 99, 'extra forward')
    require(not trainer.model.training, 'dropout/train mode active')
    with trainer._amp_context():
        require(torch.is_grad_enabled() and torch.is_autocast_enabled('cuda') and
                torch.get_autocast_dtype('cuda') == torch.bfloat16, 'not gradient-enabled bf16 eager context')
        output = trainer.model(batches['A']['x'])
        actual = {}
        for arm in ('A', 'B'):
            actual[arm] = captured_loss(compute_loss, output, batches[arm], kwargs[arm])
    require(all(bool(torch.isfinite(v).all()) for v in output.values() if isinstance(v, torch.Tensor)), 'nonfinite forward')
    result_a, terms_a, fallback_a, family_a = actual['A']
    result_b, terms_b, fallback_b, family_b = actual['B']
    require(torch.equal(family_a['P'], family_b['P']) and torch.equal(family_a['U'], family_b['U']), 'common loss families differ')
    values = [family_a['P'], family_a['U'], family_a['W'], family_a['C'], family_b['W'], family_b['C']]
    totals = [gradients(actual[arm][0]['total'], parameters, torch, np, counter, retain=True)[0]
              for arm in ('A', 'B')] if total_control else []
    vectors, coverage = [], {}
    for index_family, (name, value) in enumerate(zip(FAMILIES, values)):
        vector, info = gradients(value, parameters, torch, np, counter, retain=index_family < 5)
        vectors.append(vector); coverage[name] = info
    matrix = gram(vectors, np)
    decomposition = {}
    for arm, terms, fallback in [('A', terms_a, fallback_a), ('B', terms_b, fallback_b)]:
        decomposition[arm] = {'components': [{'name': n, 'weight': w, 'mean': float(t.detach()),
            'mask': m.detach().cpu().tolist(), 'mask_sum': float(m.detach().sum()),
            'masked_mean_denominator': float(m.detach().sum().clamp_min(1)),
            'recovered_masked_numerator': float((t.detach() * m.detach().sum().clamp_min(1))),
            'zero_mask': not bool(m.detach().any())} for n, w, t, m in terms],
            'fallback': {k: {'signature': observer.tensors_signature({k: v})[k],
                             'values': v.detach().cpu().tolist()} if isinstance(v, torch.Tensor) else v
                         for k, v in fallback.items()}, 'actual_total': float(actual[arm][0]['total'].detach()),
            'grouped_scalar_residual': float((sum(actual[arm][3].values()) - actual[arm][0]['total']).detach())}
    residual = {}
    for arm, target, wi, ci in zip(('A', 'B'), totals, (2, 4), (3, 5)):
        combined = vectors[0] + vectors[1] + vectors[wi] + vectors[ci]
        residual[arm] = {'family_sum_vs_actual_total_relative_L2': relative_error(combined, target, np),
                        'gate': False, 'scope': 'BF16 finite cotangent accumulation need not be bitexact linear'}
    del totals
    by_group = {}
    offset = 0
    groups = defaultdict(list)
    for row in mapping:
        stop = offset + row['numel']; aliases = row['aliases']
        layers = {'.'.join(n.split('.')[:2]) for n in aliases
                  if n.startswith(('blocks.', 'layer_smolgens.'))}
        layer = 'tied_across_layers' if len(layers) > 1 else next(iter(layers), row['name'].split('.')[0])
        groups['layer/' + layer].append((offset, stop)); groups['route/' + row['route']].append((offset, stop)); offset = stop
    for name, slices in groups.items():
        part = sum((gram([v[a:b] for v in vectors], np) for a, b in slices), np.zeros((6, 6)))
        by_group[name] = part.tolist()
    norm_sq = {name: float(matrix[j, j]) for j, name in enumerate(FAMILIES)}
    report = {'host_signatures': hosts, 'forward': observer.tensors_signature(output),
              'components': decomposition, 'coverage': coverage, 'gram': matrix.tolist(),
              'norm_squared': norm_sq,
              'W_over_C_norm_ratio': {arm: math.sqrt(norm_sq['W_'+arm] / norm_sq['C_'+arm])
                                     if norm_sq['C_'+arm] > 1e-24 else None for arm in ('A','B')},
              'cosines': cosines(matrix), 'group_gram': by_group, 'total_gradient_controls': residual}
    return report, vectors

        all_states, physical_map = {}, None
        for item in p['states']:
            reports, mapping, refs = run_state(item, p, parts, dataset, selected, observer, inputs, counter, out)
            identity_map = [{k:v for k,v in row.items() if k not in ('lr','state_keys_at_map')} for row in mapping]
            require(physical_map is None or identity_map == physical_map, 'trunk membership differs across states')
            physical_map = identity_map; all_states[item['label']] = reports
            gc.collect(); torch.cuda.empty_cache(); require(all(x() is None for x in refs), 'prior model still resident')
        require(counter['forwards'] == 99 and counter['gradients'] == 600 and
                counter['autograd_calls'] <= 600, 'fixed delivery count differs')
        values = [row['cosines']['d'] for row in all_states['DONOR']]
        primary = {'status':'INCONCLUSIVE_UNDEFINED_PRIMARY', 'values':values}
        if all(x is not None and math.isfinite(x) for x in values):
            rng = np.random.default_rng(2026111501); data = np.asarray(values, np.float64)
            means = data[rng.integers(0,32,size=(10000,32))].mean(axis=1)
            primary = {'status':'COMPLETE_TRAIN_GEOMETRY_ONLY', 'values':values, 'mean':float(data.mean()),
                       'bootstrap95':np.quantile(means,[.025,.975]).tolist(), 'seed':2026111501}
        strata = {kind: {'games':sum(row['identity']['end_class'] == kind for row in all_states['DONOR']),
                         'values':[row['cosines']['d'] for row in all_states['DONOR'] if row['identity']['end_class'] == kind],
                         'diagnostic_only':True} for kind in ('NATURAL','STRICT_TB_ADJUDICATED')}
        write(out/'readout.json', {'status':primary['status'], 'source_revision':REV, 'primary':primary,
                                  'states':all_states, 'counters':counter, 'optimizer_steps':0,
                                  'descriptive_endclass_strata':strata,
                                  'HOLD_loaded':False, 'raw_Euclidean_not_optimizer_or_strength':True,
                                  'peak_cuda_allocated':torch.cuda.max_memory_allocated()})
    except BaseException as error:
        write(out/'failure.json', {'status':'UNQUALIFIED_OR_INCONCLUSIVE_NO_RETRY', 'error':repr(error), 'counters':counter})
        raise
    finally:
        signal.alarm(0)


if __name__ == '__main__':
    main()
