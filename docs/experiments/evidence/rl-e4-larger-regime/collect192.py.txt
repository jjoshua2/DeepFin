def require(ok, message):
    if not ok:
        raise ValueError(message)


def digest(path):
    value = hashlib.sha256()
    with Path(path).open('rb') as handle:
        for block in iter(lambda: handle.read(8 * 1024 * 1024), b''):
            value.update(block)
    return value.hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), allow_nan=False).encode()


def write_json(path, value):
    with Path(path).open('x') as handle:
        json.dump(value, handle, indent=2, allow_nan=False)
        handle.write('\n')
        handle.flush()
        os.fsync(handle.fileno())

def freeze_tree(path, *, seal_root=True):
    require(not path.is_symlink() and all(not p.is_symlink() for p in path.rglob('*')), 'output tree contains symlink')
    files = sorted(p for p in path.rglob('*') if p.is_file())
    inventory = []
    for item in files:
        require(not item.is_symlink(), 'output contains symlink')
        inventory.append({'path': str(item.relative_to(path)), 'sha256': digest(item), 'bytes': item.stat().st_size})
        item.chmod(0o444)
        with item.open('rb') as handle:
            os.fsync(handle.fileno())
    for directory in sorted((p for p in path.rglob('*') if p.is_dir()), reverse=True):
        directory.chmod(0o555)
        sync_dir(directory)
    if seal_root:
        path.chmod(0o555)
    sync_dir(path)
    return inventory


def sync_dir(path):
    fd = os.open(path, os.O_RDONLY | os.O_DIRECTORY)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)


def unqualified_sample_evidence(samples, planned, callback_id):
    """Failure-only bounded hashes, never eligible raw targets or replay arrays."""
    from dataclasses import fields
    import numpy as np

    max_rows, max_field_bytes, max_total_bytes = 450, 1024 * 1024, 64 * 1024 * 1024
    used = 0
    rows = []
    for i, sample in enumerate(samples[:max_rows]):
        values = {}
        for descriptor in fields(sample):
            value = getattr(sample, descriptor.name)
            if isinstance(value, np.ndarray):
                item = {'shape': list(value.shape), 'dtype': value.dtype.str, 'bytes': value.nbytes}
                if value.dtype.hasobject or value.nbytes > max_field_bytes or used + value.nbytes > max_total_bytes:
                    item['hash_status'] = 'UNHASHED_OBJECT_OR_FIXED_BYTE_CAP'
                else:
                    item['sha256_C_order'] = hashlib.sha256(value.tobytes(order='C')).hexdigest()
                    used += value.nbytes
                values[descriptor.name] = item
            else:
                # Scalar repr preserves diagnostic NaN/inf without invalid JSON.
                values[descriptor.name] = {'python_type': type(value).__name__, 'repr': repr(value)[:256]}
        rows.append({'row_index': i, 'physical_row_id':
                     (planned['physical_game_id'] + ':' + str(i)) if planned is not None else None,
                     'sample_fields': values})
    return {'status': 'UNQUALIFIED_CALLBACK_SAMPLE_HASH_EVIDENCE_ONLY',
            'callback_id': callback_id, 'available_samples': len(samples), 'rows': rows,
            'max_rows': max_rows, 'max_field_bytes': max_field_bytes, 'max_total_hashed_bytes': max_total_bytes,
            'hashed_bytes': used, 'unhashed_rows': max(0, len(samples) - max_rows),
            'eligible_raw_archive_or_replay': False,
            'limits': 'Partial diagnostic hashes only if fixed caps apply; no proof, learning, sample validity or complete quarantine claim.'}

def attach_mirror_input_ids(rows, stored_x, *, input_history_encoding):
    from chess_anti_engine.replay.augment import _mirror_x_batch
    import numpy as np
    require(stored_x.shape == (len(rows), 175, 8, 8) and stored_x.dtype == np.dtype('float16'), 'unqualified175F16 mirror layout')
    mirrored = _mirror_x_batch(stored_x, np.ones(len(rows), dtype=bool), input_history_encoding=input_history_encoding)
    for i, row in enumerate(rows):
        raw = np.ascontiguousarray(stored_x[i], dtype='<f2')
        require(hashlib.sha256(raw.tobytes()).hexdigest() == row['stored_input_float16_sha256'], 'raw stored input identity differs before mirror')
        row['stored_input_float16_mirror_sha256'] = hashlib.sha256(np.ascontiguousarray(mirrored[i], dtype='<f2').tobytes()).hexdigest()


def exposure_check(rows, exposure):
    bad = [row['physical_row_id'] for row in rows if row['history_sha256'] in exposure['histories'] or (row['stored_input_float16_sha256'] in exposure['inputs'] or row['stored_input_float16_mirror_sha256'] in exposure['inputs'])]
    return {'known_overlap_rows': len(bad), 'physical_row_ids': bad, 'coverage_scope': exposure['coverage_scope'], 'donor_pretraining_exposure': 'UNKNOWN'}


def select_holdout_root(rows, planned, exposure, salt):
    # Row list is already proven full-history, actual neural-policy and nonSFrefute.
    eligible = [row for row in rows if 8 <= row['move_offset'] <= 80 and row['history_sha256'] not in exposure['histories'] and row['stored_input_float16_sha256'] not in exposure['inputs'] and row['stored_input_float16_mirror_sha256'] not in exposure['inputs']]
    if not eligible:
        return None
    root = min(eligible, key=lambda row: hashlib.sha256((salt + ':' + row['physical_row_id']).encode('utf-8')).digest())
    return {'physical_game_id': planned['physical_game_id'], **root, 'selection': 'fixed_offset8..80_hash_min_scoreblind', 'donor_pretraining_exposure': 'UNKNOWN'}

def validate_game(audit, samples, model, game_cfg, planned, seen, *, proof, verifier):
    import chess
    import numpy as np
    from chess_anti_engine.encoding import encode_cboard_for_model, model_encoding_kwargs, rep_fix
    from chess_anti_engine.encoding._lc0_ext import CBoard
    from chess_anti_engine.moves import index_to_move_strict, move_to_index_for_encoding
    from chess_anti_engine.selfplay.finalize import _stable_game_id

    start = audit.start
    verified = require_verified(proof, audit=audit)
    enc = model_encoding_kwargs(model)
    require(enc == {'input_history_encoding': game_cfg.input_history_encoding, 'input_extra_features': game_cfg.input_extra_features}, 'producer/model encoding mismatch')
    require(rep_fix.current() is game_cfg.history_rep_fix is True, 'producer runtime repetition flag mismatch')
    require(start.start_ordinal == planned['start_ordinal'] and 0 <= start.slot < 16, 'unplanned ordinal or invalid recycled slot')
    require(start.root_fen == planned['root_fen'] and list(start.opening_moves) == planned['opening_moves'], 'start history mismatch')
    require(start.is_selfplay, 'curriculum game outside pilot')
    board = chess.Board(start.root_fen)
    for uci in start.opening_moves:
        board.push_uci(uci)
    require(board.fen() == start.start_fen and board.ply() == start.start_ply, 'start state mismatch')
    before = []
    played_uci = []
    for action in audit.played_move_indices:
        before.append(board.copy())
        move = index_to_move_strict(action, board)
        require(move in board.legal_moves, 'illegal played action')
        played_uci.append(move.uci())
        board.push(move)
    require(board.fen() == chess.Board(audit.final_fen).fen() and board.ply() == audit.total_plies, 'trace/native final board mismatch')
    require(audit.game_id == _stable_game_id(start_fen=start.start_fen, opening_source=start.opening_source, move_trace=','.join(str(m) for m in audit.played_move_indices), result=audit.result, total_plies_played=audit.total_plies), 'producer game content identity mismatch')
    records = {}
    for rec in audit.records:
        require(0 <= rec.move_offset < len(before), 'record offset outside actual moves')
        position = before[rec.move_offset]
        require(position.ply() == rec.ply_index and position.turn == rec.pov_color, 'record POV/ply mismatch')
        require(not rec.is_sf_refute_opp and rec.has_policy, 'non-network/fast record outside pilot')
        require(rec.ply_index not in records, 'duplicate raw record ply')
        records[rec.ply_index] = (rec, position)
    bank = planned['bank']
    # Physical-game banks are preassigned; shared trajectory patterns are counted, not silently discarded.
    rows = []
    for row_index, sample in enumerate(samples):
        require(sample.game_id == audit.game_id and sample.ply_index in records, 'sample/raw record join mismatch')
        rec, position = records[sample.ply_index]
        require(sample.has_policy and sample.is_network_turn is True, 'sample not actual full neural turn')
        require(sample.input_history_encoding == model.input_history_encoding and sample.history_rep_fix is True, 'sample encoding contract mismatch')
        expected = encode_cboard_for_model(model, CBoard.from_board(position))
        actual = np.asarray(sample.x)
        require(expected.shape == actual.shape == (175, 8, 8) and np.isfinite(actual).all(), 'sample input shape/finiteness')
        require(np.array_equal(expected, actual), 'full-history/model-aware raw175plane input mismatch')
        legal = np.zeros(1858, dtype=np.uint8)
        for move in position.legal_moves:
            legal[move_to_index_for_encoding(move, position, policy_encoding='lc0_1858')] = 1
        require(sample.legal_mask is not None and np.array_equal(np.asarray(sample.legal_mask).astype(bool), legal.astype(bool)), 'sample legal mask mismatch')
        policy = np.asarray(sample.policy_target)
        require(policy.shape == (1858,) and np.isfinite(policy).all() and (policy >= 0).all(), 'policy target invalid')
        require(abs(float(policy.sum()) - 1) < 0.002 and not np.any(policy[legal == 0]), 'policy target normalization/legality')
        verifier.verify_policy_sample(proof, sample, position, game_cfg.policy_encoding)
        history_id = hashlib.sha256(canonical([start.root_fen, list(start.opening_moves) + played_uci[:rec.move_offset]])).hexdigest()
        input_id = hashlib.sha256(actual.astype(np.float16).tobytes()).hexdigest()
        # Root eligibility is filtered against ALL completed TRAIN inputs later.
        seen[bank]['histories'].add(history_id)
        seen[bank]['inputs'].add(input_id)
        rows.append({'physical_row_id': planned['physical_game_id'] + ':' + str(row_index), 'ply': sample.ply_index, 'move_offset': rec.move_offset, 'history_sha256': history_id, 'stored_input_float16_sha256': input_id})
    require(len({x['ply'] for x in rows}) == len(rows), 'duplicate physical-row ply')
    seen[bank]['games'].add(audit.game_id)
    return {'rows': rows, 'raw_records': len(records), 'retained_rows': len(samples),
            'resolved_outcome_eligible': verified['resolved'], 'scientific_outcome_class': verified['scientific_class'],
            'verified_outcome': verified, 'verified_outcome_sha256': proof.proof_sha256,
            'played_plies': len(played_uci), 'full_history_validated': True,
            'policy_declared_retained_rows_reprobed': True,
            'encoding': {**enc, 'history_rep_fix': game_cfg.history_rep_fix, 'runtime_rep_fix': rep_fix.current(), 'record_lc0_root_input': game_cfg.record_lc0_root_input, 'record_relations': game_cfg.record_relations, 'raw175plane_equality': True, 'stored_hash_precision': 'float16'}}


def playing_overlap_gate(completed, protocol):
    """ALL628 selected start raw/mirror IDs against ALL192 retained raw/mirror IDs."""
    p = Path(protocol['playing_start_identities_path'])
    require(digest(p) == protocol['playing_start_identities_sha256'], 'playing metadata changed')
    selected = json.loads(p.read_text())
    require(len(selected) == 820, 'not fixed820 metadata')
    playing = [row for row in selected if row['role'] in ('screen', 'confirmation')]
    require(len(playing) == 628 and sum(row['role'] == 'screen' for row in playing) == 128, 'not fixed128+500 playing starts')
    retained = {row[key] for game in completed.values() for row in game['validation']['rows']
                for key in ('stored_input_float16_sha256', 'stored_input_float16_mirror_sha256')}
    collisions = [{'candidate_id': row['candidate_id'], 'role': row['role'],
                   'matching_hashes': sorted({row['stored175F16_input_sha256'], row['stored175F16_mirror_sha256']} & retained)}
                  for row in playing if {row['stored175F16_input_sha256'], row['stored175F16_mirror_sha256']} & retained]
    return {'status': 'PASS_ALL628_VS_ALL192_RETAINED_RAW_MIRROR' if not collisions else 'INCONCLUSIVE_PLAYING_START_COLLISION_NO_LEARNING',
            'qualified': not collisions, 'all192_physical_attempts': len(completed), 'playing_starts': len(playing),
            'playing_metadata_sha256': protocol['playing_start_identities_sha256'], 'collisions': collisions,
            'no_replacement_or_new_seed': True, 'unknown': 'unstored/search/staging/pretraining/dropout variants not indexed'}
