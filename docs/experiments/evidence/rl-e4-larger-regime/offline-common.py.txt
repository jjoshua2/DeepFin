def require(ok, message):
    if not ok:
        raise ValueError(message)


def digest(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()


def checked_json(path, expected):
    require(digest(path) == expected, "bound metadata changed: " + str(path))
    return json.loads(Path(path).read_text())


def write_json(path, value):
    with Path(path).open("x") as stream:
        json.dump(value, stream, indent=2, sort_keys=True, allow_nan=False)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())


def module(name, path, expected):
    require(digest(path) == expected, "qualified helper changed: " + str(path))
    sys.path.insert(0, str(Path(path).parent))
    spec = importlib.util.spec_from_file_location(name, path)
    result = importlib.util.module_from_spec(spec)
    sys.modules[name] = result
    spec.loader.exec_module(result)
    return result

def hash_input(value):
    x = np.asarray(value)
    require(
        x.shape == (175, 8, 8) and x.dtype == np.dtype("<f2") and np.isfinite(x).all(),
        "root input not finite175F16",
    )
    return hashlib.sha256(np.ascontiguousarray(x).tobytes()).hexdigest()


def verify_owner(protocol, *, resource, seconds, affinity, cuda):
    args = (
        Path("/proc/" + str(os.getppid()) + "/cmdline")
        .read_bytes()
        .decode()
        .split("\0")
    )
    require(
        protocol["owner_path"] in args
        and digest(protocol["owner_path"]) == protocol["owner_sha256"],
        "wrong finite offline owner",
    )
    require(
        "--resource" in args
        and args[args.index("--resource") + 1] == resource
        and "--seconds" in args
        and args[args.index("--seconds") + 1] == str(seconds),
        "offline owner resource/ceiling differs",
    )
    require(
        sorted(os.sched_getaffinity(0)) == affinity
        and os.environ.get("CUDA_VISIBLE_DEVICES") == cuda
        and __debug__
        and os.environ.get("OMP_NUM_THREADS")
        == os.environ.get("MKL_NUM_THREADS")
        == "2",
        "offline visibility/affinity/assertions/thread contract differs",
    )
