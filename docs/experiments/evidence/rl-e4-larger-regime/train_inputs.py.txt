def require(ok, message):
    if not ok:
        raise ValueError(message)


def digest(path):
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def checked_json(path, expected):
    require(isinstance(expected, str) and len(expected) == 64 and
            all(c in "0123456789abcdef" for c in expected), "metadata hash UNBOUND")
    require(digest(path) == expected, "bound metadata changed: " + str(path))
    return json.loads(Path(path).read_text())


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()


def proc_birth(pid):
    try:
        return int(Path(f"/proc/{pid}/stat").read_text().rsplit(")", 1)[1].split()[19])
    except FileNotFoundError:
        return None


def freeze_tree(path):
    require(not path.is_symlink() and all(not p.is_symlink() for p in path.rglob("*")), "symlink output")
    for p in path.rglob("*"):
        if p.is_file():
            p.chmod(0o444)
            with p.open("rb") as stream:
                os.fsync(stream.fileno())
    for p in sorted([p for p in path.rglob("*") if p.is_dir()], reverse=True) + [path]:
        p.chmod(0o555)
        fd = os.open(p, os.O_RDONLY | os.O_DIRECTORY)
        try:
            os.fsync(fd)
        finally:
            os.close(fd)

def inventory(game, entries):
    require(not game.is_symlink() and all(not p.is_symlink() for p in game.rglob("*")), "symlink TRAIN directory")
    declared = set()
    for item in entries:
        rel = Path(item["path"])
        require(not rel.is_absolute() and ".." not in rel.parts and str(rel) not in declared, "unsafe/duplicate inventory")
        declared.add(str(rel))
        path = game / rel
        require(path.is_file() and not path.is_symlink() and
                all(not p.is_symlink() for p in path.parents) and
                path.stat().st_mode & 0o222 == 0, "mutable/nonregular TRAIN artifact")
        require(path.stat().st_size == item["bytes"] and digest(path) == item["sha256"], "TRAIN inventory changed")
    require({str(p.relative_to(game)) for p in game.rglob("*") if p.is_file()} == declared,
            "TRAIN inventory not complete")

def record_positions(audit):
    import chess
    from chess_anti_engine.moves import index_to_move_strict
    board = chess.Board(audit["start"]["root_fen"])
    for token in audit["start"]["opening_moves"]:
        board.push_uci(token)
    positions = {}
    wanted = {r["move_offset"]: r["ply_index"] for r in audit["records"]}
    for offset, action in enumerate(audit["played_move_indices"]):
        if offset in wanted:
            positions[wanted[offset]] = board.copy(stack=True)
        board.push(index_to_move_strict(action, board))
    return positions


def validate_loaded_game(arrays, raw, meta, rows, audit, proof, verifier, game):
    from chess_anti_engine.encoding.cboard_encode import encode_cboard
    from chess_anti_engine.encoding._lc0_ext import CBoard
    from chess_anti_engine.moves import move_to_index_for_encoding
    from chess_anti_engine.selfplay.finalize import _result_to_wdl
    from chess_anti_engine.train.targets import categorical_target_value, hlgauss_target
    from verified_outcome import require_verified
    from value_bundle import validate_stored_control
    verified = require_verified(proof, audit=audit, metadata=meta)
    n = len(rows)
    require(0 < n <= 450 and arrays["x"].shape == (n, 175, 8, 8) and
            arrays["x"].dtype == np.float16 and len(raw["physical_row_id"]) == n and
            raw["physical_row_id"].tolist() == [r["physical_row_id"] for r in rows], "raw/storage physical rows differ")
    for flag in ("has_game_id", "has_ply_index", "has_is_network_turn", "has_legal_mask"):
        require(flag in arrays and arrays[flag].shape == (n,) and np.all(arrays[flag] == 1),
                "serialized retained identity/NET/legal availability missing: " + flag)
    require(arrays["game_id"].shape == arrays["ply_index"].shape == arrays["is_network_turn"].shape == (n,) and
            np.all(arrays["game_id"] == audit["game_id"]) and np.all(arrays["is_network_turn"] == 1),
            "serialized game identity or NET turn differs from audit")
    for key in ("is_sf_refute", "is_sf_refute_opp"):
        if key in arrays:
            require(arrays[key].shape == (n,) and np.all(arrays[key] == 0), "serialized SF-refute row outside E4")
    require(set(raw) == {"physical_row_id", "search_wdl_raw", "categorical_raw", "has_search_raw", "has_categorical_raw", "outcome_wdl", "pov_white"},
            "raw archive fields changed")
    for key, shape in (("search_wdl_raw", (n, 3)), ("categorical_raw", (n, game.categorical_bins))):
        x = raw[key]
        require(x.dtype == np.float32 and x.shape == shape and np.isfinite(x).all() and
                (x >= 0).all() and np.all(np.abs(x.sum(axis=1) - 1) < 1e-5), "rawF32 simplex/dtype changed")
    for key in ("has_search_raw", "has_categorical_raw"):
        require(raw[key].shape == (n,) and raw[key].dtype == np.uint8 and np.all(raw[key] == 1), "missing raw control")
    require(raw["outcome_wdl"].shape == raw["pov_white"].shape == (n,) and
            raw["outcome_wdl"].dtype == np.int8 and raw["pov_white"].dtype == np.uint8, "raw outcome/POV layout changed")
    positions = record_positions(audit)
    records = {r["ply_index"]: r for r in audit["records"]}
    require(len({r["ply"] for r in rows}) == n, "duplicate retained physical ply")
    from chess_anti_engine.replay.augment import _mirror_x_batch
    mirrored = _mirror_x_batch(arrays["x"], np.ones(n, dtype=bool), input_history_encoding=game.input_history_encoding)
    for i, row in enumerate(rows):
        position = positions[row["ply"]]
        require(arrays["ply_index"][i] == row["ply"] == records[row["ply"]]["ply_index"] and
                records[row["ply"]]["has_policy"] is True and
                records[row["ply"]]["is_sf_refute_opp"] is False,
                "serialized/raw retained ply or actual NET record provenance differs")
        require(records[row["ply"]]["move_offset"] == row["move_offset"] and
                hashlib.sha256(canonical([position.root().fen(), [m.uci() for m in position.move_stack]])).hexdigest() == row["history_sha256"],
                "retained full-history/ply/offset identity differs")
        require(int(raw["pov_white"][i]) == int(position.turn) and
                int(raw["outcome_wdl"][i]) == _result_to_wdl(verified["result"], pov_white=position.turn), "resolved per-row POV mismatch")
        expected = encode_cboard(CBoard.from_board(position), input_history_encoding=game.input_history_encoding,
                                 input_extra_features=game.input_extra_features).astype(np.float16)
        require(np.array_equal(expected, arrays["x"][i]), "replayed175F16 input differs")
        require(hashlib.sha256(arrays["x"][i].astype('<f2').tobytes()).hexdigest() == row["stored_input_float16_sha256"] and
                hashlib.sha256(mirrored[i].astype('<f2').tobytes()).hexdigest() == row["stored_input_float16_mirror_sha256"], "raw/mirror row identity differs")
        legal = np.zeros(1858, np.uint8)
        for move in position.legal_moves:
            legal[move_to_index_for_encoding(move, position, policy_encoding=game.policy_encoding)] = 1
        policy = arrays["policy_target"][i]
        require(np.array_equal(arrays["legal_mask"][i], legal) and arrays["has_policy"][i] == 1 and
                np.isfinite(policy).all() and (policy >= 0).all() and abs(float(policy.sum(dtype=np.float32)) - 1) < .002 and
                not np.any(policy[legal == 0]), "serialized legal policy mismatch")
        verifier.verify_policy_sample(proof, SimpleNamespace(ply_index=row["ply"], policy_target=policy), position, game.policy_encoding)
        q = 1 if raw["outcome_wdl"][i] == 0 else (0 if raw["outcome_wdl"][i] == 1 else -1)
        control = hlgauss_target(categorical_target_value(q, None, blend_frac=0, search_wdl=raw["search_wdl_raw"][i], search_blend_frac=1),
                                num_bins=game.categorical_bins, sigma=game.hlgauss_sigma)
        require(control.tobytes() == raw["categorical_raw"][i].tobytes(), "raw original categorical control differs")
    validate_stored_control(raw, arrays, proof=proof, metadata=meta)


def load_research_train(binding, parts):
    """Reverify ALL128 proofs before deserializing a single TRAIN archive/shard."""
    from chess_anti_engine.replay.shard import load_shard_arrays
    from chess_anti_engine.selfplay.config import GameConfig
    from scripts.offline_replay_epoch import _concat
    from verified_outcome import VerifiedOutcomeVerifier, require_verified
    _gen, producer, _flat, tc, _kwargs, _model_cfg = parts
    readout, planned = cohort_metadata(binding, producer)
    game = GameConfig(**producer["training_game_config"])
    verifier = VerifiedOutcomeVerifier(path=producer["tablebase_path"],
        asset_inventory_sha256=producer["tablebase_asset_inventory_sha256"], collection_uuid=RUN, source_revision=SOURCE)
    prepared = []
    try:
        for g in planned:
            item = readout["completed"][str(g["start_ordinal"])]
            directory = Path(item["path"])
            inventory(directory, item["inventory"])  # Stream hashes only; no array deserialization.
            validation = json.loads((directory / "validation.json").read_text())
            require(validation == item["validation"] and validation["usable_shard"] is True and
                    validation["resolved_outcome_eligible"] is True and
                    validation["scientific_outcome_class"] in ("NATURAL", "STRICT_TB_ADJUDICATED") and
                    validation["raw_value_control"]["all_rows_raw_control_proven"] is True, "unusable/censored TRAIN game")
            observed = json.loads((directory / "provenance.json").read_text())
            require(observed["planned"] == g, "raw planned TRAIN identity changed")
            proof = verifier.verify(observed["actual"], g, game)
            require_verified(proof, audit=observed["actual"])
            stored_proof = json.loads((directory / "verified-outcome.json").read_text())
            require(stored_proof == {"verification": proof.metadata(), "proof_sha256": proof.proof_sha256}, "fresh/serialized outcome proof differs")
            rows = validation["rows"]
            require(0 < len(rows) <= 450 and len(rows) == validation["retained_rows"] == proof.payload["retained_rows"] and
                    [r["physical_row_id"] for r in rows] == [g["physical_game_id"] + ":" + str(i) for i in range(len(rows))],
                    "not complete ordered physical TRAIN rows")
            prepared.append((g, directory, validation, observed["actual"], proof))
        require(len(prepared) == 128 and len({g[0]["physical_game_id"] for g in prepared}) == 128, "not ALL128 fresh TRAIN proofs")
        chunks, raw_chunks, ids, games, segments = [], [], [], [], []
        for g, directory, validation, audit, proof in prepared:
            arrays, stored_meta = load_shard_arrays(directory / "shard", lazy=False, validate=True)
            require(stored_meta["run_id"] == RUN and stored_meta["model_sha256"] == producer["checkpoint_sha256"], "TRAIN shard source/donor differs")
            with np.load(directory / "raw-values.npz", allow_pickle=False) as archive:
                raw = {key: archive[key].copy() for key in archive.files}
            meta = json.loads((directory / "raw-values.json").read_text())
            require(meta == validation["raw_value_control"] and meta["all_rows_raw_control_proven"] is True,
                    "raw metadata differs from admitted game")
            validate_loaded_game(arrays, raw, meta, validation["rows"], audit, proof, verifier, game)
            segments.append({"start": len(ids), "stop": len(ids) + len(raw["physical_row_id"]), "metadata": meta, "proof": proof})
            chunks.append(arrays); raw_chunks.append(raw); ids.extend(raw["physical_row_id"].tolist()); games.append(g["physical_game_id"])
        require(0 < len(ids) <= 128 * 450 and len(ids) == len(set(ids)), "TRAIN row bound/uniqueness")
        arrays = _concat(chunks)
        raw = {key: np.concatenate([x[key] for x in raw_chunks], axis=0) for key in raw_chunks[0]}
        metadata = {"segments": segments, "categorical_bins": game.categorical_bins, "hlgauss_sigma": game.hlgauss_sigma}
        validate_train_stored_control(raw, arrays, metadata=metadata)
        return arrays, ids, games, raw, metadata, {"scope": "accepted128wholeTRAINonly_fresh_verified_outcomes",
            "cohort_binding": binding, "heldout_loaded": False, "games": games, "rows": len(ids),
            "fresh_proof_sha256": [x["proof"].proof_sha256 for x in segments],
            "classes": [x["proof"].payload["scientific_class"] for x in segments]}
    finally:
        verifier.close()


def segments(raw, arrays, metadata):
    from scripts.offline_replay_epoch import _slice_arrays
    from verified_outcome import require_verified
    require(len(metadata["segments"]) == 128, "not128 proof segments")
    cursor = 0
    for segment in metadata["segments"]:
        start, stop = segment["start"], segment["stop"]
        require(start == cursor and start < stop <= len(raw["physical_row_id"]), "proof segment gap/overlap")
        require_verified(segment["proof"], metadata=segment["metadata"])
        ix = np.arange(start, stop)
        yield {k: v[start:stop] for k, v in raw.items()}, (_slice_arrays(arrays, ix) if arrays is not None else None), segment
        cursor = stop
    require(cursor == len(raw["physical_row_id"]), "unproven trailing TRAIN rows")


def validate_train_stored_control(raw, arrays, *, metadata):
    from value_bundle import validate_stored_control
    for values, stored, segment in segments(raw, arrays, metadata):
        validate_stored_control(values, stored, proof=segment["proof"], metadata=segment["metadata"])


def categorical_for_train_arm(raw, *, arm, metadata):
    from value_bundle import categorical_for_arm
    return np.concatenate([categorical_for_arm(values, arm=arm, metadata=segment["metadata"], proof=segment["proof"])
                           for values, _stored, segment in segments(raw, None, metadata)], axis=0)
