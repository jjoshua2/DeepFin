import Base
import ./Spec.bend as S
import ../board/Spec.bend as B
import ../castling/Route.bend as R
import ../castle_emission/Spec.bend as E
import ../../../legal_probe/Chess.bend as Chess

law transit_preserves_moving_king_singleton:
  for +b: Chess.Board
  for +r: R.Route
  for good: {B.valid(b) == True{} : Bool}
  for +one: {S.owned(b,Chess.get_turn(b)) == U64.bit(U32.to_nat(R.src(r))) : U64}
  {S.owned(S.intermediate(b,r),Chess.get_turn(b)) == U64.bit(U32.to_nat(S.transit(r))) : U64}

law castle_preserves_moving_king_singleton:
  for +b: Chess.Board
  for +r: R.Route
  for +good: {B.valid(b) == True{} : Bool}
  for +one: {S.owned(b,Chess.get_turn(b)) == U64.bit(U32.to_nat(R.src(r))) : U64}
  for fresh: {B.fresh_mask(b,R.rook(r)) == True{} : Bool}
  {S.owned(S.final(b,r),Chess.get_turn(b)) == R.target(r) : U64}

law guard_supplies_castling_singleton:
  for +b: Chess.Board
  for +ks: Bool
  for good: {B.valid(b) == True{} : Bool}
  for one: {S.owned(b,Chess.get_turn(b)) == U64.bit(U32.to_nat(R.src(E.route(b,ks)))) : U64}
  for guard: {E.guard(b,ks) == True{} : Bool}
  {S.owned(Chess.make_move(b,E.move(b,ks)),Chess.get_turn(b)) == R.target(E.route(b,ks)) : U64}

law guarded_checks_target_original_side_king:
  for table: Array<U64>
  for +b: Chess.Board
  for +ks: Bool
  for +stage: S.Stage
  for good: {B.valid(b) == True{} : Bool}
  for one: {S.owned(b,Chess.get_turn(b)) == U64.bit(U32.to_nat(R.src(E.route(b,ks)))) : U64}
  for guard: {E.guard(b,ks) == True{} : Bool}
  {Chess.in_check(table,S.at(b,E.route(b,ks),stage),Chess.get_turn(b)) ==
    Chess.attacked(table,S.at(b,E.route(b,ks),stage),S.square(E.route(b,ks),stage),U32.xor(Chess.get_turn(b),1)) : Array<U64> & Bool}

law move_flips_valid_side:
  for +b: Chess.Board
  for +m: Chess.Ply
  for +white: Bool
  for valid: {Chess.get_turn(b) == Bool.to_u32(white) : U32}
  {Chess.get_turn(Chess.make_move(b,m)) == Bool.to_u32(Bool.not(white)) : U32}
