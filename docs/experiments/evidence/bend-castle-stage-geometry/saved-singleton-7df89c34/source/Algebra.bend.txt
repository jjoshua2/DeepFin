# Structural bit-vector algebra. No board or occupancy enumeration.
import Base

def join_word(+n: Nat,+ah: Bool,+at: Word(n),+bh: Bool,+bt: Word(n),h: {ah == bh : Bool},t: {at == bt : Word(n)}) ->
  {WCon{ah,at} == WCon{bh,bt} : Word(1n+n)}:
  Equal.trans(Word(1n+n),WCon{ah,at},WCon{bh,at},WCon{bh,bt},
    Equal.cong(Bool,Word(1n+n),x => WCon{x,at},ah,bh,h),
    Equal.cong(Word(n),Word(1n+n),x => WCon{bh,x},at,bt,t))

def join64(+al: Word(32n),+ah: Word(32n),+bl: Word(32n),+bh: Word(32n),l: {al == bl : Word(32n)},h: {ah == bh : Word(32n)}) ->
  {U64{U32{al},U32{ah}} == U64{U32{bl},U32{bh}} : U64}:
  Equal.trans(U64,U64{U32{al},U32{ah}},U64{U32{bl},U32{ah}},U64{U32{bl},U32{bh}},
    Equal.cong(Word(32n),U64,x => U64{U32{x},U32{ah}},al,bl,l),
    Equal.cong(Word(32n),U64,x => U64{U32{bl},U32{x}},ah,bh,h))

def projection_bit(+k: Bool,+w: Bool,+m: Bool,+t: Bool,+r: Bool) ->
  {Bool.and(Bool.or(Bool.and(k,Bool.not(m)),t),Bool.or(Bool.and(w,Bool.not(m)),Bool.or(t,r))) == Bool.or(Bool.or(Bool.and(Bool.and(k,w),Bool.not(m)),t),Bool.and(Bool.and(k,Bool.not(m)),r)) : Bool}:

    case False{} False{} False{} False{} False{}: {==}

def projection_word(+n: Nat,+k: Word(n),+w: Word(n),+m: Word(n),+t: Word(n),+r: Word(n)) ->
  {Word.and(n,Word.or(n,Word.and(n,k,Word.not(n,m)),t),Word.or(n,Word.and(n,w,Word.not(n,m)),Word.or(n,t,r))) == Word.or(n,Word.or(n,Word.and(n,Word.and(n,k,w),Word.not(n,m)),t),Word.and(n,Word.and(n,k,Word.not(n,m)),r)) : Word(n)}:
  match n:
    case 0n: {==}
    case 1n+p:
      match k w m t r:
        case WCon{+kh,+kt} WCon{+wh,+wt} WCon{+mh,+mt} WCon{+th,+tt} WCon{+rh,+rt}:
          join_word(p,Bool.and(Bool.or(Bool.and(kh,Bool.not(mh)),th),Bool.or(Bool.and(wh,Bool.not(mh)),Bool.or(th,rh))),Word.and(p,Word.or(p,Word.and(p,kt,Word.not(p,mt)),tt),Word.or(p,Word.and(p,wt,Word.not(p,mt)),Word.or(p,tt,rt))),Bool.or(Bool.or(Bool.and(Bool.and(kh,wh),Bool.not(mh)),th),Bool.and(Bool.and(kh,Bool.not(mh)),rh)),Word.or(p,Word.or(p,Word.and(p,Word.and(p,kt,wt),Word.not(p,mt)),tt),Word.and(p,Word.and(p,kt,Word.not(p,mt)),rt)),projection_bit(kh,wh,mh,th,rh),projection_word(p,kt,wt,mt,tt,rt))

def projection(+k: U64,+w: U64,+m: U64,+t: U64,+r: U64) ->
  {U64.and(U64.or(U64.and(k,U64.not(m)),t),U64.or(U64.and(w,U64.not(m)),U64.or(t,r))) == U64.or(U64.or(U64.and(U64.and(k,w),U64.not(m)),t),U64.and(U64.and(k,U64.not(m)),r)) : U64}:
  match k w m t r:
    case U64{U32{+kl},U32{+kh}} U64{U32{+wl},U32{+wh}} U64{U32{+ml},U32{+mh}} U64{U32{+tl},U32{+th}} U64{U32{+rl},U32{+rh}}:
      join64(Word.and(32n,Word.or(32n,Word.and(32n,kl,Word.not(32n,ml)),tl),Word.or(32n,Word.and(32n,wl,Word.not(32n,ml)),Word.or(32n,tl,rl))),Word.and(32n,Word.or(32n,Word.and(32n,kh,Word.not(32n,mh)),th),Word.or(32n,Word.and(32n,wh,Word.not(32n,mh)),Word.or(32n,th,rh))),Word.or(32n,Word.or(32n,Word.and(32n,Word.and(32n,kl,wl),Word.not(32n,ml)),tl),Word.and(32n,Word.and(32n,kl,Word.not(32n,ml)),rl)),Word.or(32n,Word.or(32n,Word.and(32n,Word.and(32n,kh,wh),Word.not(32n,mh)),th),Word.and(32n,Word.and(32n,kh,Word.not(32n,mh)),rh)),projection_word(32n,kl,wl,ml,tl,rl),projection_word(32n,kh,wh,mh,th,rh))

def prune_bit(+k: Bool,+m: Bool,+r: Bool) ->
  {Bool.and(Bool.and(k,Bool.not(m)),r) == Bool.and(Bool.and(k,r),Bool.not(m)) : Bool}:
  match k m r:
    case False{} False{} False{}: {==}
    case False{} False{} True{}: {==}
    case False{} True{} False{}: {==}
    case False{} True{} True{}: {==}
    case True{} False{} False{}: {==}
    case True{} False{} True{}: {==}
    case True{} True{} False{}: {==}
    case True{} True{} True{}: {==}

def prune_word(+n: Nat,+k: Word(n),+m: Word(n),+r: Word(n)) ->
  {Word.and(n,Word.and(n,k,Word.not(n,m)),r) == Word.and(n,Word.and(n,k,r),Word.not(n,m)) : Word(n)}:
  match n:
    case 0n: {==}
    case 1n+p:
      match k m r:
        case WCon{+kh,+kt} WCon{+mh,+mt} WCon{+rh,+rt}:
          join_word(p,Bool.and(Bool.and(kh,Bool.not(mh)),rh),Word.and(p,Word.and(p,kt,Word.not(p,mt)),rt),Bool.and(Bool.and(kh,rh),Bool.not(mh)),Word.and(p,Word.and(p,kt,rt),Word.not(p,mt)),prune_bit(kh,mh,rh),prune_word(p,kt,mt,rt))

def prune(+k: U64,+m: U64,+r: U64) ->
  {U64.and(U64.and(k,U64.not(m)),r) == U64.and(U64.and(k,r),U64.not(m)) : U64}:
  match k m r:
    case U64{U32{+kl},U32{+kh}} U64{U32{+ml},U32{+mh}} U64{U32{+rl},U32{+rh}}:
      join64(Word.and(32n,Word.and(32n,kl,Word.not(32n,ml)),rl),Word.and(32n,Word.and(32n,kh,Word.not(32n,mh)),rh),Word.and(32n,Word.and(32n,kl,rl),Word.not(32n,ml)),Word.and(32n,Word.and(32n,kh,rh),Word.not(32n,mh)),prune_word(32n,kl,ml,rl),prune_word(32n,kh,mh,rh))
