def rook_from(+dst: U32) -> U32:
  Bool.pick(U32,U32.is_eq(U32.and(dst,7),6),U32.inc(dst),U32.sub(dst,2))

def rook_to(src: U32,dst: U32) -> U64:
  U64.bit(U32.to_nat(U32.div(U32.add(src,dst),2)))

def mask(+src: U32,+dst: U32) -> U64:
  U64.or(U64.or(U64.bit(U32.to_nat(src)),U64.bit(U32.to_nat(dst))),U64.or(U64.bit(U32.to_nat(dst)),U64.bit(U32.to_nat(rook_from(dst)))))
