HERE = Path(__file__).resolve().parent


def require(ok, message):
    if not ok:
        raise ValueError(message)


def digest(path):
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def checked(ref):
    require(set(ref) == {"path", "sha256"} and "UNBOUND" not in ref["path"], "identity remains UNBOUND")
    require(digest(ref["path"]) == ref["sha256"], "bound identity changed")
    return json.loads(Path(ref["path"]).read_text())


def write(path, value):
    with Path(path).open("x") as stream:
        json.dump(value, stream, indent=2, sort_keys=True, allow_nan=False)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())

    ap.add_argument("--execute", action="store_true")
    ap.add_argument("--manifest-sha256", required=True)
    ap.add_argument("--bindings", required=True)
    ap.add_argument("--bindings-sha256", required=True)
    args = ap.parse_args()

    require(digest(HERE / "manifest.json") == args.manifest_sha256, "draw helper manifest changed")
    m = json.loads((HERE / "manifest.json").read_text())
    require(digest(__file__) == m["helper_sha256"], "draw helper changed")
    require(digest(args.bindings) == args.bindings_sha256, "late bindings changed")
    bindings = json.loads(Path(args.bindings).read_text())

    require(m["owner_path"] in parent and digest(m["owner_path"]) == m["owner_sha256"] and
            "--seconds" in parent and parent[parent.index("--seconds") + 1] == "300" and
            "--resource" in parent and parent[parent.index("--resource") + 1] == "cpu", "wrong finite owner")
    require(sorted(os.sched_getaffinity(0)) == [20, 22] and os.environ.get("CUDA_VISIBLE_DEVICES") == "" and
            os.environ.get("OMP_NUM_THREADS") == os.environ.get("MKL_NUM_THREADS") == "2" and
            __debug__ and os.environ.get("PYTHONOPTIMIZE", "0") == "0", "CPU resource contract changed")

        require(digest(ref["path"]) == ref["sha256"], "fixed authority changed")
    protocol = checked(m["paired_protocol_template"])
    manifest = checked(m["paired_manifest_template"])
    science, publication, producer, _pm, cohort = [checked(bindings[key]) for key in
        ("scientific", "publication", "producer_protocol", "producer_manifest", "cohort")]

            science["cohort"]["uuid"] == protocol["run_uuid"] and
            publication["status"] == "MAIN_PUBLISHED_FROZEN_E2TB_SCIENCE" and
            publication["scientific_sha256"] == bindings["scientific"]["sha256"], "science is not frozen and published")

        "manifest": {"path": str(HERE / "manifest.json"), "sha256": digest(HERE / "manifest.json")},
        "identity_only_inputs_template": m["inputs_template"]}
    require(science["components"]["draw_plan_binder"] == expected_component, "science does not pin this fixed draw binder")
    require(producer["scientific_preregistration_sha256"] == bindings["scientific"]["sha256"] and
            producer["collection_uuid"] == protocol["run_uuid"], "producer scientific identity differs")
    require(cohort["producer_protocol_sha256"] == bindings["producer_protocol"]["sha256"] and

        protocol[stem + "_sha256"] = bindings[key]["sha256"]
    require(protocol["execution_binding_complete"] is protocol["execution_admitted"] is False and
            protocol["steps"] == 40 and protocol["batch_size"] == 256 and protocol["draw_seed"] == 2026111401,
            "fixed dose or non-execution scope changed")
    for name, expected in manifest["helpers"].items():
        require(digest(Path(m["TRAIN_code_directory"]) / name) == expected, "TRAIN helper changed")
    output = Path(m["output_root"])
    require(not output.exists() and not Path(protocol["draw_plan_path"]).exists(), "draw output already exists")

            attempts[name] += 1
