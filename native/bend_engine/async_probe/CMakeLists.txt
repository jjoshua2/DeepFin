cmake_minimum_required(VERSION 3.18)
project(deepfin_async_model_probe LANGUAGES CXX)
if(NOT DEFINED BEND_CONTRACT_DIR)
  message(FATAL_ERROR "BEND_CONTRACT_DIR is required; generate a CPU-singleton header with bind_model.js")
endif()
find_package(Torch REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(native-async-model-probe model_probe.cpp ../standalone/async_model.cpp ../standalone/model_bridge.cpp)
set_property(TARGET native-async-model-probe PROPERTY CXX_STANDARD 20)
set_property(TARGET native-async-model-probe PROPERTY CXX_STANDARD_REQUIRED ON)
target_include_directories(native-async-model-probe PRIVATE "${BEND_CONTRACT_DIR}")
target_compile_options(native-async-model-probe PRIVATE -O1 -ffp-contract=off)
target_link_libraries(native-async-model-probe PRIVATE "${TORCH_LIBRARIES}" Threads::Threads OpenSSL::Crypto m)
