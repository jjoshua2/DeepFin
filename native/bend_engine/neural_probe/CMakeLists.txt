cmake_minimum_required(VERSION 3.18 FATAL_ERROR)
project(deepfin_bend_neural_worker LANGUAGES CXX)
find_package(Torch REQUIRED)
find_package(Threads REQUIRED)
add_executable(aoti_worker aoti_worker.cpp)
set_property(TARGET aoti_worker PROPERTY CXX_STANDARD 20)
set_property(TARGET aoti_worker PROPERTY CXX_STANDARD_REQUIRED ON)
target_compile_options(aoti_worker PRIVATE -O2)
target_link_libraries(aoti_worker PRIVATE "${TORCH_LIBRARIES}" Threads::Threads)
