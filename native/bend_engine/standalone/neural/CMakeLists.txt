cmake_minimum_required(VERSION 3.18 FATAL_ERROR)
project(deepfin_bend_native_model LANGUAGES C CXX)
find_package(Torch REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
if(NOT DEFINED BEND_GENERATED_C OR NOT DEFINED MODEL_CONFIG_DIR)
  message(FATAL_ERROR "BEND_GENERATED_C and MODEL_CONFIG_DIR are required")
endif()
add_executable(deepfin-bend-neural "${BEND_GENERATED_C}" model.cpp)
set_property(TARGET deepfin-bend-neural PROPERTY C_STANDARD 11)
set_property(TARGET deepfin-bend-neural PROPERTY CXX_STANDARD 20)
set_property(TARGET deepfin-bend-neural PROPERTY CXX_STANDARD_REQUIRED ON)
target_include_directories(deepfin-bend-neural PRIVATE "${MODEL_CONFIG_DIR}")
target_compile_options(deepfin-bend-neural PRIVATE -O1 -ffp-contract=off
  $<$<COMPILE_LANGUAGE:C>:-Werror=shift-count-overflow>)
if(DEFINED BEND_MODE AND BEND_MODE STREQUAL "portable")
  target_compile_definitions(deepfin-bend-neural PRIVATE BEND_U64_PORTABLE)
elseif(DEFINED BEND_MODE AND BEND_MODE STREQUAL "native")
  target_compile_options(deepfin-bend-neural PRIVATE -march=native)
elseif(DEFINED BEND_MODE AND BEND_MODE STREQUAL "ubsan")
  target_compile_options(deepfin-bend-neural PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
  target_link_options(deepfin-bend-neural PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
endif()
target_link_libraries(deepfin-bend-neural PRIVATE "${TORCH_LIBRARIES}" Threads::Threads OpenSSL::Crypto m)
