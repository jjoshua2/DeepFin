cmake_minimum_required(VERSION 3.18)
project(deepfin_bend_native_model LANGUAGES C CXX)
if(NOT DEFINED BEND_GENERATED_C OR NOT DEFINED BEND_CONTRACT_DIR)
  message(FATAL_ERROR "BEND_GENERATED_C and BEND_CONTRACT_DIR are required")
endif()
find_package(Torch REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
if(NOT DEFINED BEND_TARGET_NAME)
  set(BEND_TARGET_NAME deepfin-bend-neural)
endif()
add_executable(${BEND_TARGET_NAME} "${BEND_GENERATED_C}" model_bridge.cpp)
set_property(TARGET ${BEND_TARGET_NAME} PROPERTY C_STANDARD 11)
set_property(TARGET ${BEND_TARGET_NAME} PROPERTY CXX_STANDARD 20)
set_property(TARGET ${BEND_TARGET_NAME} PROPERTY CXX_STANDARD_REQUIRED ON)
target_compile_definitions(${BEND_TARGET_NAME} PRIVATE DEEPFIN_BEND_NATIVE_MODEL)
target_include_directories(${BEND_TARGET_NAME} PRIVATE "${BEND_CONTRACT_DIR}")
target_compile_options(${BEND_TARGET_NAME} PRIVATE -O1 -ffp-contract=off)
target_link_libraries(${BEND_TARGET_NAME} PRIVATE "${TORCH_LIBRARIES}" Threads::Threads OpenSSL::Crypto m)

# Cheap tensor-layout failure controls, requested only by explicit qualification.
if(BEND_BATCH_TESTS)
  add_executable(batch-output-contract ../batch_backend/output_contract_test.cpp)
  set_property(TARGET batch-output-contract PROPERTY CXX_STANDARD 20)
  target_link_libraries(batch-output-contract PRIVATE "${TORCH_LIBRARIES}")
endif()
