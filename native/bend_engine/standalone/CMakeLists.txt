cmake_minimum_required(VERSION 3.18)
project(deepfin_bend_native_model LANGUAGES C CXX)
if(NOT DEFINED BEND_GENERATED_C OR NOT DEFINED BEND_CONTRACT_DIR)
  message(FATAL_ERROR "BEND_GENERATED_C and BEND_CONTRACT_DIR are required")
endif()
find_package(Torch REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(deepfin-bend-neural "${BEND_GENERATED_C}" model_bridge.cpp)
set_property(TARGET deepfin-bend-neural PROPERTY C_STANDARD 11)
set_property(TARGET deepfin-bend-neural PROPERTY CXX_STANDARD 20)
set_property(TARGET deepfin-bend-neural PROPERTY CXX_STANDARD_REQUIRED ON)
target_compile_definitions(deepfin-bend-neural PRIVATE DEEPFIN_BEND_NATIVE_MODEL)
target_include_directories(deepfin-bend-neural PRIVATE "${BEND_CONTRACT_DIR}")
target_compile_options(deepfin-bend-neural PRIVATE -O1 -ffp-contract=off)
target_link_libraries(deepfin-bend-neural PRIVATE "${TORCH_LIBRARIES}" Threads::Threads OpenSSL::Crypto m)
